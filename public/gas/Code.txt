/**
 * Serves the HTML Page on Web App Access
 */
function doGet(e) {
  var template = HtmlService.createTemplateFromFile("index");
  return template.evaluate()
    .setTitle("Doctoral Student Portfolio")
    .addMetaTag("viewport", "width=device-width, initial-scale=1.0")
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

/**
 * Returns the active Spreadsheet. If not set, triggers setupDatabase().
 */
function getSpreadsheet() {
  var ssId = SPREADSHEET_ID_PROP;
  return SpreadsheetApp.openById(ssId);
}

/**
 * Gets the configured Google Drive Folder ID
 */
function getDriveFolder() {
  var folderId = DRIVE_FOLDER_ID_PROP;
  return DriveApp.getFolderById(folderId);
}

/**
 * Setup Database and Sheets Schema
 * Does NOT clear existing data. Safely appends missing sheets and columns.
 */
function setupDatabase() {
  var ssId = SPREADSHEET_ID_PROP;
  var ss = SpreadsheetApp.openById(ssId);

  // Create Google Drive folder for evidence uploads
  var folderId = DRIVE_FOLDER_ID_PROP;
  var folder = DriveApp.getFolderById(folderId);

  // Configure Sheets & Headers
  var schemas = {
    "Settings": ["SettingKey", "SettingValue", "Description", "Example", "Options", "UpdatedAt", "UpdatedBy"],
    "Users": ["UserID", "Email", "Password", "Role", "Prefix", "FirstName", "LastName", "FullName", "StudentID", "Program", "Faculty", "University", "AdmissionYear", "ExpectedGraduationYear", "MajorAdvisorID", "CoAdvisorIDs", "Position", "Affiliation", "Phone", "LineID", "ResearchInterests", "ORCID", "PhotoURL", "Status", "CreatedAt", "UpdatedAt"],
    "PortfolioRecords": ["RecordID", "StudentUserID", "SectionNo", "SectionTitle", "SubsectionNo", "SubsectionTitle", "RecordType", "Field1", "Field2", "Field3", "Field4", "Field5", "Field6", "Field7", "LongText", "Status", "EvidenceIDs", "CreatedBy", "CreatedAt", "UpdatedBy", "UpdatedAt"],
    "StudentProfile": ["ProfileID", "StudentUserID", "FullName", "ContactInformation", "CurrentPositionAffiliation", "ResearchInterests", "ORCID", "GoalsForDoctoralStudy", "DevelopmentPlan", "EnglishReflection", "ResearchExperienceReflection", "NetworkingReflection", "CommunicationReflection", "AcademicGrowthReflection", "ResearchIdentityReflection", "ChallengesReflection", "TransformationReflection", "ShortTermCareerGoals", "LongTermCareerAspirations", "PreparationNeeded", "UpdatedAt", "UpdatedBy"],
    "Dissertation": ["DissertationID", "StudentUserID", "TopicDevelopment", "Title", "BackgroundSignificance", "ResearchProblem", "Objectives", "ResearchQuestionsHypotheses", "ConceptualFramework", "MethodologyOverview", "EthicsApplicationDate", "EthicsApprovalDate", "ApprovalNumber", "Amendments", "DataManagementNotes", "ChallengesSolutions", "UpdatedAt", "UpdatedBy"],
    "ResearchHours": ["HourID", "StudentUserID", "Date", "ResearchActivity", "WorkDescription", "Hours", "SupervisorAdvisor", "EvidenceIDs", "CreatedBy", "CreatedAt", "UpdatedAt"],
    "CompetencyAssessment": ["AssessmentID", "StudentUserID", "Competency", "Level", "EvidenceRemarks", "ReviewYear", "CreatedBy", "CreatedAt", "UpdatedAt"],
    "AdvisorComments": ["CommentID", "StudentUserID", "AdvisorUserID", "ReviewYear", "CommentText", "Recommendation", "Status", "CreatedAt", "UpdatedAt"],
    "Endorsements": ["EndorsementID", "StudentUserID", "Role", "AdvisorUserID", "AdvisorName", "SignatureText", "SignatureDate", "Status", "CreatedAt", "UpdatedAt"],
    "Evidence": ["EvidenceID", "StudentUserID", "RelatedRecordID", "RelatedSection", "FileName", "FileURL", "FileID", "MimeType", "UploadedBy", "UploadedAt", "Description"],
    "Notifications": ["NotificationID", "SenderUserID", "ReceiverUserID", "Title", "Message", "FileName", "FileURL", "IsRead", "CreatedAt", "ReadAt"],
    "ChatMessages": ["MessageID", "ThreadID", "SenderUserID", "ReceiverUserID", "StudentUserID", "MessageText", "FileName", "FileURL", "IsRead", "CreatedAt"],
    "ActivityLog": ["LogID", "UserID", "Action", "Detail", "CreatedAt"]
  };

  for (var sheetName in schemas) {
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
    }
    
    // Check or create headers
    var lastCol = sheet.getLastColumn();
    var headers = schemas[sheetName];
    
    if (lastCol === 0) {
      sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
      sheet.setFrozenRows(1);
      
      // Styling header row (Dark Red or Golden depending on sheet)
      var headerRange = sheet.getRange(1, 1, 1, headers.length);
      headerRange.setBackground("#B91C1C").setFontColor("#FFFFFF").setFontWeight("bold");
      sheet.autoResizeColumns(1, headers.length);
    } else {
      // Add missing columns if schema changed
      var existingHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
      for (var i = 0; i < headers.length; i++) {
        if (existingHeaders.indexOf(headers[i]) === -1) {
          sheet.insertColumnAfter(sheet.getLastColumn());
          sheet.getRange(1, sheet.getLastColumn()).setValue(headers[i]);
        }
      }
    }

    // Set formatting rules for Plain text columns
    if (sheetName === "Users") {
      // Format Phone (19th col), StudentID (9th col), LineID (20th col) as Text
      sheet.getRange("I:I").setNumberFormat("@"); // StudentID
      sheet.getRange("S:S").setNumberFormat("@"); // Phone
      sheet.getRange("T:T").setNumberFormat("@"); // LineID
    }
  }

  // Remove default sheet if exists
  var defaultSheet = ss.getSheetByName("Sheet1");
  if (defaultSheet) {
    try {
      ss.deleteSheet(defaultSheet);
    } catch(e) {}
  }

  return {
    spreadsheetId: ss.getId(),
    spreadsheetUrl: ss.getUrl(),
    folderId: folder.getId(),
    folderUrl: folder.getUrl()
  };
}

 * Reset Sample Data (removes SYSTEM_SAMPLE rows only)

  var ss = getSpreadsheet();
  var sheets = ["Settings", "Users", "PortfolioRecords", "StudentProfile", "Dissertation", "ResearchHours", "CompetencyAssessment", "AdvisorComments", "Endorsements", "Evidence", "Notifications", "ChatMessages", "ActivityLog"];

    var sheet = ss.getSheetByName(name);

    var data = sheet.getRange(1, 1, sheet.getLastRow(), sheet.getLastColumn()).getValues();
    var headers = data[0];
    var createdByCol = headers.indexOf("CreatedBy");
    var updatedByCol = headers.indexOf("UpdatedBy");
    
    // Fallback for Users sheet / Settings sheet which uses different columns or no CreatedBy
    var keysToKeep = [];
    keysToKeep.push(headers);
    
    for (var r = 1; r < data.length; r++) {
      var row = data[r];
      var isSample = false;
      
      if (createdByCol !== -1 && (row[createdByCol] === "SYSTEM_SAMPLE" || row[createdByCol] === "U001")) {
        isSample = true;
      }
      if (updatedByCol !== -1 && (row[updatedByCol] === "SYSTEM_SAMPLE")) {
        isSample = true;
      }
      // Settings sheet condition
      if (name === "Settings" && row[6] === "SYSTEM_SAMPLE") {
        isSample = true;
      }
      // Users sheet condition
      if (name === "Users" && (row[0] === "U001" || row[0] === "U002" || row[0] === "U003" || row[0] === "U004" || row[0] === "U005")) {
        isSample = true;
      }
      
      if (!isSample) {
        keysToKeep.push(row);
      }
    }
    
    sheet.clearContents();
    if (keysToKeep.length > 0) {
      sheet.getRange(1, 1, keysToKeep.length, headers.length).setValues(keysToKeep);
    }
  });

  return { success: true, message: "Sample data cleaned up successfully." };
}

/**
 * Handle user Login
 */
function login(email, password) {
  var ss = getSpreadsheet();
  var sheet = ss.getSheetByName("Users");
  var rows = getSheetDataJSON(sheet);

  for (var i = 0; i < rows.length; i++) {
    var u = rows[i];
    if (String(u.Email).toLowerCase().trim() === String(email).toLowerCase().trim()) {
      if (String(u.Password) === String(password)) {
        if (u.Status !== "Active") {
          throw new Error("This account is currently suspended.");
        }
        // Strip sensitive Password out before passing to client state
        delete u.Password;
        return u;
      }
    }
  }
  throw new Error("Invalid email or password.");
}

/**
 * Retrieve entire system state for client caching on load
 */
function getAppData(userId) {
  var ss = getSpreadsheet();
  var userSheet = ss.getSheetByName("Users");
  var users = getSheetDataJSON(userSheet);
  
  var currentUser = users.find(function(u) { return u.UserID === userId; });
  if (!currentUser) throw new Error("User session not found.");
  
  var role = currentUser.Role;

  // Retrieve raw tables
  var settings = getSheetDataJSON(ss.getSheetByName("Settings"));
  var records = getSheetDataJSON(ss.getSheetByName("PortfolioRecords"));
  var profiles = getSheetDataJSON(ss.getSheetByName("StudentProfile"));
  var dissertations = getSheetDataJSON(ss.getSheetByName("Dissertation"));
  var researchHours = getSheetDataJSON(ss.getSheetByName("ResearchHours"));
  var competencyAssessments = getSheetDataJSON(ss.getSheetByName("CompetencyAssessment"));
  var comments = getSheetDataJSON(ss.getSheetByName("AdvisorComments"));
  var endorsements = getSheetDataJSON(ss.getSheetByName("Endorsements"));
  var evidence = getSheetDataJSON(ss.getSheetByName("Evidence"));
  var notifications = getSheetDataJSON(ss.getSheetByName("Notifications"));
  var chatMessages = getSheetDataJSON(ss.getSheetByName("ChatMessages"));

  // Strip password for safety
  users.forEach(function(u) { delete u.Password; });

  // Filtering based on roles (Students, Advisor, CoAdvisor, Admin)
  if (role === "Student") {
    records = records.filter(function(r) { return r.StudentUserID === userId; });
    profiles = profiles.filter(function(p) { return p.StudentUserID === userId; });
    dissertations = dissertations.filter(function(d) { return d.StudentUserID === userId; });
    researchHours = researchHours.filter(function(h) { return h.StudentUserID === userId; });
    competencyAssessments = competencyAssessments.filter(function(a) { return a.StudentUserID === userId; });
    comments = comments.filter(function(c) { return c.StudentUserID === userId; });
    endorsements = endorsements.filter(function(e) { return e.StudentUserID === userId; });
    evidence = evidence.filter(function(e) { return e.StudentUserID === userId; });
    notifications = notifications.filter(function(n) { return n.ReceiverUserID === userId; });
    chatMessages = chatMessages.filter(function(m) {
      return m.StudentUserID === userId || m.SenderUserID === userId || m.ReceiverUserID === userId;
    });
  } else if (role === "Advisor") {
    var advisedStudentIds = users
      .filter(function(u) { return u.MajorAdvisorID === userId; })
      .map(function(u) { return u.UserID; });

    records = records.filter(function(r) { return advisedStudentIds.indexOf(r.StudentUserID) !== -1; });
    profiles = profiles.filter(function(p) { return advisedStudentIds.indexOf(p.StudentUserID) !== -1; });
    dissertations = dissertations.filter(function(d) { return advisedStudentIds.indexOf(d.StudentUserID) !== -1; });
    researchHours = researchHours.filter(function(h) { return advisedStudentIds.indexOf(h.StudentUserID) !== -1; });
    competencyAssessments = competencyAssessments.filter(function(a) { return advisedStudentIds.indexOf(a.StudentUserID) !== -1; });
    comments = comments.filter(function(c) { 
      return advisedStudentIds.indexOf(c.StudentUserID) !== -1 || c.AdvisorUserID === userId; 
    });
    endorsements = endorsements.filter(function(e) { 
      return advisedStudentIds.indexOf(e.StudentUserID) !== -1 || e.AdvisorUserID === userId; 
    });
    evidence = evidence.filter(function(e) { return advisedStudentIds.indexOf(e.StudentUserID) !== -1; });
    notifications = notifications.filter(function(n) {
      return n.SenderUserID === userId || advisedStudentIds.indexOf(n.ReceiverUserID) !== -1;
    });
    chatMessages = chatMessages.filter(function(m) {
      return m.SenderUserID === userId || m.ReceiverUserID === userId || advisedStudentIds.indexOf(m.StudentUserID) !== -1;
    });
  } else if (role === "CoAdvisor") {
    var linkedStudentIds = users
      .filter(function(u) { 
        if(!u.CoAdvisorIDs) return false;
        var coAdvs = u.CoAdvisorIDs.split(",");
        return coAdvs.indexOf(userId) !== -1; 
      })
      .map(function(u) { return u.UserID; });

    records = records.filter(function(r) { return linkedStudentIds.indexOf(r.StudentUserID) !== -1; });
    profiles = profiles.filter(function(p) { return linkedStudentIds.indexOf(p.StudentUserID) !== -1; });
    dissertations = dissertations.filter(function(d) { return linkedStudentIds.indexOf(d.StudentUserID) !== -1; });
    researchHours = researchHours.filter(function(h) { return linkedStudentIds.indexOf(h.StudentUserID) !== -1; });
    competencyAssessments = competencyAssessments.filter(function(a) { return linkedStudentIds.indexOf(a.StudentUserID) !== -1; });
    comments = comments.filter(function(c) { return linkedStudentIds.indexOf(c.StudentUserID) !== -1; });
    endorsements = endorsements.filter(function(e) { return linkedStudentIds.indexOf(e.StudentUserID) !== -1; });
    evidence = evidence.filter(function(e) { return linkedStudentIds.indexOf(e.StudentUserID) !== -1; });
    notifications = notifications.filter(function(n) {
      return n.SenderUserID === userId || linkedStudentIds.indexOf(n.ReceiverUserID) !== -1;
    });
    chatMessages = chatMessages.filter(function(m) {
      return m.SenderUserID === userId || m.ReceiverUserID === userId || linkedStudentIds.indexOf(m.StudentUserID) !== -1;
    });
  }

  return {
    users: users,
    portfolioRecords: records,
    studentProfiles: profiles,
    dissertations: dissertations,
    researchHours: researchHours,
    competencyAssessments: competencyAssessments,
    advisorComments: comments,
    endorsements: endorsements,
    evidence: evidence,
    notifications: notifications,
    chatMessages: chatMessages,
    settings: settings
  };
}

/**

  var file = folder.createFile(blob);

  file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    evidenceId,
    metadata.StudentUserID,

    fileName,
    file.getUrl(),
    file.getId(),
    mimeType,
    metadata.UploadedBy || metadata.StudentUserID,
    new Date(),
    metadata.Description || ""
  ];

  sheet.appendRow(rowValues);

  return {
    success: true,
    evidenceId: evidenceId,
    fileName: fileName,
    fileUrl: file.getUrl()
  };
}

    var incoming = settingsList[s];
    for (var i = 0; i < data.length; i++) {
      if (data[i].SettingKey === incoming.SettingKey) {
        sheet.getRange(i + 2, 2).setValue(incoming.SettingValue); // SettingValue col B
        sheet.getRange(i + 2, 6).setValue(new Date()); // UpdatedAt col F
        break;
      }
    }
  }
  return { success: true };
}

/**
 * Converts any Google Sheet's 2D grid into a clean array of JSON objects
 * Handles dates, booleans, and empty cells elegantly
 */
function getSheetDataJSON(sheet) {
  if (!sheet || sheet.getLastRow() <= 1) return [];
  
  var range = sheet.getRange(1, 1, sheet.getLastRow(), sheet.getLastColumn());
  var values = range.getValues();
  var headers = values[0];
  var list = [];
  
  for (var r = 1; r < values.length; r++) {
    var row = values[r];
    var obj = {};
    var isEmptyRow = true;
    
    for (var c = 0; c < headers.length; c++) {
      var val = row[c];
      
      // Clean up stringified numbers with leading zero indicators if present
      if (typeof val === "string" && val.charAt(0) === "'") {
        val = val.substring(1);
      }
      
      obj[headers[c]] = val;
      if (val !== "" && val !== null && val !== undefined) {
        isEmptyRow = false;
      }
    }
    if (!isEmptyRow) {
      list.push(obj);
    }
  }
  return list;
}
